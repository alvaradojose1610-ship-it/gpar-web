    };
    this.measure();

    this.mt = setInterval(this.measure, 500);
  }
  componentWillUnmount(){ window.removeEventListener("resize", this.measure); clearInterval(this.mt); }
  top(){ try{ window.scrollTo(0,0); }catch(e){} }
  nav(view, cat){ this.setState({view, cat, sub:"Todos", cartOpen:false}); this.top(); }

    const go = () => { const el = document.getElementById(id); if(el) window.scrollTo(0, el.getBoundingClientRect().top + window.scrollY - ((this.state.headerH || 72) + 8)); };
    if(this.state.view !== "home"){ this.setState({view:"home", cat:null, cartOpen:false}); setTimeout(go, 80); }
    else go();
