<script type="text/x-dc" data-dc-script>
class Component extends DCLogic {
  state = { sound: false };
  bgRef = React.createRef();
  mainRef = React.createRef();
  componentDidMount() {
    const v = this.mainRef.current, bg = this.bgRef.current;
    if (bg) { bg.muted = true; bg.play().catch(() => {}); }
    if (v) {
      v.muted = false;
      v.play().then(() => this.setState({ sound: true })).catch(() => { v.muted = true; v.play().catch(() => {}); });
    }
    this.firstClick = (e) => {
      document.removeEventListener('pointerdown', this.firstClick, true);
      if (e.target.closest && e.target.closest('button')) return;
      if (!this.state.sound) this.setSound(true);
    };
    document.addEventListener('pointerdown', this.firstClick, true);
  }
  componentWillUnmount() { document.removeEventListener('pointerdown', this.firstClick, true); }
  setSound(on) {
    const v = this.mainRef.current;
    if (v) { v.muted = !on; if (on) v.play().catch(() => {}); }
    this.setState({ sound: on });
  }
  renderVals() {
    return {
      bgRef: this.bgRef, mainRef: this.mainRef,
      toggle: () => this.setSound(!this.state.sound),
      label: this.state.sound ? 'Silenciar' : 'Activar sonido'
    };
  }
}
</script>
